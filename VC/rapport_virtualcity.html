📖 1. Introduction

VirtualCity est un projet de simulation de gestion de ville développé en C++ avec le framework Qt. Ce simulateur permet aux utilisateurs de construire et gérer une ville virtuelle en ajoutant différents types de bâtiments, en gérant les ressources (eau, électricité, budget) et en observant l'évolution de la satisfaction des habitants au fil des cycles de simulation.

Le projet met en œuvre les concepts fondamentaux de la Programmation Orientée Objet (POO) tels que l'héritage, le polymorphisme et l'encapsulation. L'interface graphique moderne et intuitive a été développée avec Qt, offrant une expérience utilisateur fluide et agréable.

🎯 2. Objectifs du Projet

🏗️ Construction

Permettre la construction de différents types de bâtiments : Maisons, Usines et Parcs.

💰 Gestion des Ressources

Gérer le budget, l'eau, l'électricité et la pollution de la ville.

👥 Population

Suivre l'évolution de la population et sa satisfaction.

Implémenter un système de cycles avec événements aléatoires.

7.1 Classe de Base : Batiment

La classe Batiment est la classe de base abstraite pour tous les types de bâtiments :

batiment.h C++ Header 
#ifndef BATIMENT_H
#define BATIMENT_H

#include <string>
#include <iostream>
using namespace std;

class Batiment
{
protected:
    int id;
    string nom;
    string type;
    double consommationEau;
    double consommationElectricite;
    int effetSatisfaction;
    double coutConstruction;

public:
    Batiment();
    Batiment(int id, string nom, string type, 
            double eau, double elec, int satisfaction, double cout);
    virtual ~Batiment() {}

    virtual void afficherDetails();
    virtual void calculerImpactRessources(double& eau, double& electricite);

    // Getters
    int getId() { return id; }
    string getNom() { return nom; }
    string getType() { return type; }
    double getConsommationEau() { return consommationEau; }
    double getConsommationElectricite() { return consommationElectricite; }
    int getEffetSatisfaction() { return effetSatisfaction; }
    double getCoutConstruction() { return coutConstruction; }
    void setNom(const string& n) { nom = n; }
};

#endif

7.2 Classe Dérivée : Maison

La classe Maison hérite de Batiment et gère les habitants :

maison.cpp C++ 
#include "maison.h"
#include <iostream>

Maison::Maison() : Batiment() {
    capaciteHabitants = 0;
    habitantsActuels = 0;
}

Maison::Maison(int id, string nom, int capacite)
    : Batiment(id, nom, "Habitation", 10.0, 15.0, 5, 50000.0) {
    capaciteHabitants = capacite;
    habitantsActuels = 0;
}

bool Maison::ajouterHabitants(int nb) {
    if (habitantsActuels + nb <= capaciteHabitants) {
        habitantsActuels += nb;
        return true;
    }
    return false;
}

bool Maison::retirerHabitants(int nb) {
    if (habitantsActuels >= nb) {
        habitantsActuels -= nb;
        return true;
    }
    return false;
}

void Maison::afficherDetails() {
    Batiment::afficherDetails();
    cout << "Capacite Habitants: " << capaciteHabitants << endl;
    cout << "Habitants Actuels: " << habitantsActuels << endl;
}

7.5 Classe Ville

La classe Ville gère l'ensemble des bâtiments et des ressources :

ville.cpp (extrait) C++ 
void Ville::ajouterBatiment(Batiment* batiment) {
    if (budget >= batiment->getCoutConstruction()) {
        batiments.push_back(batiment);
        budget -= batiment->getCoutConstruction();

        Maison* maison = dynamic_cast<Maison*>(batiment);
        if (maison != nullptr) {
            int capacite = maison->getCapaciteHabitants();
            maison->ajouterHabitants(capacite);
        }

        calculerPopulationTotale();
        calculerPollutionTotale();
        calculerSatisfaction();
    }
}

int Ville::calculerSatisfaction() {
    int satisfactionTotale = 50;

    for (int i = 0; i < batiments.size(); i++) {
        satisfactionTotale += batiments[i]->getEffetSatisfaction();
    }

    if (pollution > 50.0) {
        satisfactionTotale -= (int)(pollution / 10.0);
    }

    if (eau < 0 || electricite < 0) {
        satisfactionTotale -= 20;
    }

    if (satisfactionTotale < 0) satisfactionTotale = 0;
    if (satisfactionTotale > 100) satisfactionTotale = 100;

    satisfaction = satisfactionTotale;
    return satisfaction;
}

7.6 Classe Simulation avec Événements

La classe Simulation gère les cycles et les événements aléatoires :

simulation.cpp (extrait) C++ 
void Simulation::initialiserEvenements() {
    Evenement e1 = {"Nuee de Pigeons", 
        "Une nuee de pigeons geants envahit la ville!", 
        -15, 0, 0, 5};
    evenements.push_back(e1);

    Evenement e2 = {"Panne de Courant", 
        "Une panne de courant fait grimper la consommation!", 
        -10, 0, 50, 0};
    evenements.push_back(e2);

    Evenement e3 = {"Festival de la Ville", 
        "Un festival ameliore le moral!", 
        25, -10, -20, -5};
    evenements.push_back(e3);

    Evenement e4 = {"Subvention Gouvernementale", 
        "La ville recoit une aide financiere!", 
        10, 50, 50, 0};
    evenements.push_back(e4);
}

Evenement Simulation::declencherEvenement() {
    if (evenements.size() == 0 || ville == nullptr) {
        return {"", "", 0, 0, 0, 0};
    }

    int index = rand() % evenements.size();
    Evenement evt = evenements[index];

    // Appliquer les impacts sur la ville
    int nouvelleSat = ville->getSatisfaction() + evt.impactSatisfaction;
    ville->setSatisfaction(nouvelleSat);

    double nouvelleEau = ville->getEau() + evt.impactEau;
    ville->setEau(nouvelleEau);

    double nouvelleElec = ville->getElectricite() + evt.impactElectricite;
    ville->setElectricite(nouvelleElec);

    dernierEvenement = evt;
    return evt;
}

MainWindow::MainWindow(QWidget *parent)
    : QMainWindow(parent)
    , ui(new Ui::MainWindow)
{
    ui->setupUi(this);

    ville = new Ville("Ma Ville Virtuelle", 500000.0);
    simulation = new Simulation(ville);
    loadFromFile(defaultSavePath());

    // Connexion des signaux/slots
    connect(ui->btnAjouterMaison, &QPushButton::clicked, 
            this, &MainWindow::ouvrirDialogMaison);
    connect(ui->btnAjouterUsine, &QPushButton::clicked, 
            this, &MainWindow::ouvrirDialogUsine);
    connect(ui->btnCycle, &QPushButton::clicked, 
            this, &MainWindow::demarrerCycle);
    connect(ui->btnSave, &QPushButton::clicked, 
            this, &MainWindow::saveToFile);

    mettreAJourAffichage();
}

void MainWindow::mettreAJourAffichage() {
    ui->labelCycle->setText("📅 Cycle: " + 
        QString::number(simulation->getCycleActuel()));
    ui->labelBudget->setText("💰 Budget: " + 
        QString::number(ville->getBudget(), 'f', 0) + " $");
    ui->labelPopulation->setText("👥 Population: " + 
        QString::number(ville->getPopulation()));
    
    int sat = ville->getSatisfaction();
    QString satEmoji = sat >= 70 ? "😊" : (sat >= 40 ? "😐" : "😫");
    ui->labelSatisfaction->setText(satEmoji + " Satisfaction: " + 
        QString::number(sat) + "%");

⚙️ 8. Fonctionnalités

8.1 Gestion des Bâtiments

  • Maison : Héberge les habitants, consomme eau (10L) et électricité (15kWh), coût 50 000$
  • Usine : Produit des ressources, génère pollution (15), consomme plus d'énergie, coût 100 000$
  • Parc : Améliore le bien-être (+10 satisfaction), faible consommation, coût 30 000$

8.2 Système d'Événements

Le jeu inclut plusieurs événements aléatoires qui peuvent survenir à chaque cycle :
